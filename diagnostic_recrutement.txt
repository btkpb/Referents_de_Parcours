    private function validateUpload(
        ?UploadedFile $file,
        array $mimeAutorises,
        string $messageErreur
    ): ?string {
        if ($file === null) {
            return null;
        }

        if (!$file->isValid()) {
            throw new \InvalidArgumentException('Un fichier transmis est invalide.');
        }

        if ($file->getSize() > self::MAX_FILE_SIZE) {
            throw new \InvalidArgumentException('Chaque fichier doit faire 10 Mo maximum.');
        }

        $mime = $file->getMimeType();

        if (!in_array($mime, $mimeAutorises, true)) {
            throw new \InvalidArgumentException($messageErreur);
        }

        return match ($mime) {
            'application/pdf' => 'pdf',
            'image/jpeg' => 'jpg',
            'image/png' => 'png',
            default => throw new \InvalidArgumentException($messageErreur),
        };
    }

    #[Route(
        '/medical/{id}/document/{type}',
        name: 'medical_document_download',
        requirements: ['id' => '\d+', 'type' => 'certificat|photo'],
        methods: ['GET']
    )]
    public function telecharger(
        VisiteMedicale $visite,
        string $type
    ): BinaryFileResponse {
        $this->assertMedicalAccess();

        $nomFichier = match ($type) {
            'certificat' => $visite->getCertificat(),
            'photo' => $visite->getPhoto(),
            default => null,
        };

        if (!$nomFichier || basename($nomFichier) !== $nomFichier) {
            throw $this->createNotFoundException('Document mÃ©dical introuvable.');
        }

        $chemin = $this->getParameter('kernel.project_dir')
            . '/var/uploads/medical/'
            . $nomFichier;

        if (!is_file($chemin)) {
            throw $this->createNotFoundException('Le fichier mÃ©dical est introuvable.');
        }

        $mime = $type === 'certificat'
            ? 'application/pdf'
            : match (strtolower(pathinfo($nomFichier, PATHINFO_EXTENSION))) {
                'jpg', 'jpeg' => 'image/jpeg',
                'png' => 'image/png',
                default => 'application/octet-stream',
            };

        $response = new BinaryFileResponse($chemin);
        $response->headers->set('Content-Type', $mime);
        $response->headers->set('X-Content-Type-Options', 'nosniff');
        $response->setContentDisposition(
            ResponseHeaderBag::DISPOSITION_ATTACHMENT,
            basename($nomFichier)
        );

        return $response;
    }
